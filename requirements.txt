# Pinned deliberately.
#
# The model is trained on the host (by Jenkins) and the resulting .pkl is
# baked into the Docker image. If the scikit-learn version inside the
# container differed from the one that created the pickle, loading it would
# raise InconsistentVersionWarning and could fail outright. Pinning keeps the
# host and the container byte-for-byte compatible.
#
# The same reasoning applies to the sentence-embedding stack: the classifier
# is trained on embeddings from these exact versions, so the container must
# produce identical embeddings at prediction time.
#
# Note: the Docker practical (Experiment 5) specifies Flask==3.0.0. We pin the
# version actually installed on this machine instead, for the same
# host/container matching reason.

# CPU-only PyTorch, the same 2.13.0 on both platforms:
#   Linux (the Docker image)  PyPI's default wheel bundles CUDA libraries
#                             worth several gigabytes that a CPU container
#                             never uses, so the "+cpu" build comes from
#                             PyTorch's own index.
#   Windows (the Jenkins host) PyPI's wheel is already CPU-only and is
#                             published as plain "2.13.0". Requiring "+cpu"
#                             here made pip re-download and reinstall torch.
# Everything else still comes from PyPI.
--extra-index-url https://download.pytorch.org/whl/cpu

torch==2.13.0+cpu; sys_platform == "linux"

sentence-transformers==5.7.0

# huggingface-hub's HTTP stack, pinned to what a clean install resolves (the
# Docker image). Jenkins installs into a Python shared with other tools; one
# of them left h11 0.16 beside an old httpcore 1.0.7 that needs h11<0.15.
# Without these pins pip "fixes" that by downgrading h11, breaking the other
# tools; with them it upgrades httpcore instead.
httpcore==1.0.9
h11==0.16.0
